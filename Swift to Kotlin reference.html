Swift → Kotlin Reference

A practical translation guide for this Android rewrite exercise.

Variables & Constants

Swift
let name: String = "Lupo"
var isFavorite: Bool = false
Kotlin
val name: String = "Lupo"
var isFavorite: Boolean = false
let → val (immutable), var → var (same). Bool → Boolean.

String Interpolation

Swift
"Hello \(name)"
Kotlin
"Hello $name"

Classes & Data Types

Reference type (class)
Swift
final class ItemViewModel: Identifiable {
    let id: UUID
    var name: String
    var isFavorite: Bool
}
Kotlin
class ItemViewModel : ViewModel() {
    // extends Android's ViewModel
}
Swift uses final class for reference types. Kotlin has no struct — use data class for plain data models.
Value type / plain model
Swift
struct Item {
    let id: UUID
    var name: String
    var isFavorite: Bool
}
Kotlin
data class Item(
    val id: UUID = UUID.randomUUID(),
    val name: String,
    val isFavorite: Boolean
)
data class auto-generates equals(), hashCode(), copy() — similar to Swift's Equatable conformance on structs.

Optionals / Nullables

Swift
var selectedItem: Item? = nil

if let item = selectedItem {
    print(item.name)
}

let name = selectedItem?.name ?? "None"
Kotlin
var selectedItem: Item? = null

selectedItem?.let { item ->
    println(item.name)
}

val name = selectedItem?.name ?: "None"
if let → ?.let { }. The nil-coalescing operator ?? → Elvis operator ?:.

Closures & Lambdas

Swift
items.sorted { $0.name < $1.name }

items.filter { $0.isFavorite }

items.map { $0.name }
Kotlin
items.sortedBy { it.name }

items.filter { it.isFavorite }

items.map { it.name }
Swift's $0 shorthand → Kotlin's it for single-parameter lambdas.

Protocols → Interfaces

Swift
protocol Identifiable {
    var id: UUID { get }
}

class Item: Identifiable, Hashable {
    let id = UUID()
}
Kotlin
interface Identifiable {
    val id: UUID
}

class Item : Identifiable, Hashable {
    override val id = UUID.randomUUID()
}

State Management (@Observable → ViewModel)

Swift
@Observable
final class ItemsListViewModel {
    private(set) var items: [Item] = []

    func addItem(named name: String) {
        items.append(Item(name: name))
    }
}
Kotlin
class ItemsListViewModel : ViewModel() {
    private val _items = mutableStateListOf<Item>()
    val items: List<Item> get() = _items

    fun addItem(name: String) {
        _items.add(Item(name = name))
    }
}
Swift mutates objects in place (@Observable reference types). Kotlin uses immutable data class — to update, replace the item in the list using copy(): _items[index] = item.copy(isFavorite = true).
